chgTimerFreq(unsigned freq) 
{                              
         unsigned lowCount,highCount; 
         unsigned long count; 

         count = 1193280 / freq; 
         lowCount = count % 256; 
         highCount = count / 256; 

         outportb(0x43,0x34); 
         outportb(0x40,lowCount); 
         outportb(0x40,highCount); 
}

// 스피드핵 만들기 & 막기 : 비엽(cronan) GPG 사이트에서 

static inline void do_8253(void) 
{ 
    __asm 
    { 
        mov     al, 34h 
        out     43h, al         ; _outp(0x43, 0x34); 
        mov     ax, bx 
        out     40h, al         ; _outp(0x40, low); 
        mov     al, ah 
        out     40h, al         ; _outp(0x40, high); 
        _emit   0xCF          ; iret 
    }; 
} 


static void __declspec(naked) Win98_8253(unsigned freq) 
{ 
    // prolog 
    __asm 
    { 
        push    ebp 
        mov     ebp, esp 
        push    ebx 
    }; 

    __asm 
    { 
        push    ecx 
        push    eax 
        sidt    [esp-2] 
        pop     eax 
        mov     [ebp-4], eax 
        mov     eax, [ebp-4] 
        mov     ecx, offset do_8253 
        mov     [ebp-4], ecx 
        mov     dl, byte ptr [ebp-1] 
        mov     [eax+28h], cl 
        mov     cl, byte ptr [ebp-2] 
        mov     [eax+29h], ch 
        mov     [eax+2Eh], cl 
        mov     [eax+2Fh], dl 
        mov     ebx, [ebp+8] 
        int     5 
    }; 

    // epilog 
    __asm 
    { 
        pop     ebx 
        mov     esp, ebp 
        pop     ebp 
        retn 
    }; 
}

static void __declspec(naked) NT_8253(unsigned freq) 
{ 
    // prolog 
    __asm 
    { 
        push    ebp 
        mov     ebp, esp 
        push    ebx 
    }; 

    __asm 
    { 
        push 0x34 
        push 0x43 
        call Outport 

        mov ebx, [ebp+8] 
        mov al, bl 
        and eax, 0xFF 
        push eax 
        push 0x40 
        call Outport 

        xor ecx, ecx 
        mov cl, bh 
        push ecx 
        push 0x40 
        call Outport 
    }; 

    // epilog 
    __asm 
    { 
        pop     ebx 
        mov     esp, ebp 
        pop     ebp 
        retn 
    }; 
}
